cmake_minimum_required(VERSION 3.22.1)
project(sanad_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
add_subdirectory(whisper.cpp whispercpp-build EXCLUDE_FROM_ALL)
add_library(sanad_whisper SHARED whisper_jni.c)
target_include_directories(sanad_whisper PRIVATE whisper.cpp/include whisper.cpp/ggml/include)
find_library(LOG_LIB log)
find_library(ANDROID_LIB android)
target_link_libraries(sanad_whisper PRIVATE whisper ${LOG_LIB} ${ANDROID_LIB})
target_compile_options(sanad_whisper PRIVATE -O3 -fvisibility=hidden -fvisibility-inlines-hidden)
